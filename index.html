# Butterworth bandpass filter + Pan-Tompkins R-peak detection
from scipy.signal import butter, filtfilt
import numpy as np

def bandpass_filter(signal, lowcut=0.5, highcut=40.0, fs=500, order=4):
    nyq = 0.5 * fs
    low, high = lowcut / nyq, highcut / nyq
    b, a = butter(order, [low, high], btype='band')
    return filtfilt(b, a, signal)

def detect_r_peaks(filtered_ecg, fs=500):
    # Differentiate → square → moving window integration
    diff = np.diff(filtered_ecg)
    squared = diff ** 2
    window = int(0.150 * fs)  # 150ms integration window
    integrated = np.convolve(squared, np.ones(window)/window, mode='same')
    threshold = np.mean(integrated) * 1.5
    return np.where(integrated > threshold)[0]
